% flight timeline: the mass-delta gate rebuilds once per recorded step while the motor burns,
% then freezes because the live mass sum repeats bit-for-bit after burnout.
\begin{tikzpicture}[x=1cm,y=1cm]
  % axes
  \draw[axis] (0,0) -- (11.6,0) node[below=2pt, font=\sffamily\scriptsize] {$t$};
  \draw[axis] (0,0) -- (0,3.2);
  \node[font=\sffamily\scriptsize, anchor=south, rotate=90] at (-0.35,1.5)
      {composite mass $m(t)$};

  % mass curve: loaded mass falling over the burn, then the constant casing mass
  \draw[line width=1.1pt, draw=qrBlue]
      (0,2.7) .. controls (1.3,2.55) and (2.9,1.5) .. (4.2,1.15) -- (11.2,1.15);
  \node[font=\sffamily\scriptsize, text=qrBlue, anchor=south west] at (0.15,2.72)
      {loaded mass};
  \node[font=\sffamily\scriptsize, text=qrBlue, anchor=south west] at (7.0,1.2)
      {empty (casing) mass --- constant};

  % ignition / burnout markers
  \draw[station] (0,0) -- (0,2.95);
  \node[font=\sffamily\scriptsize, text=qrTeal, anchor=south west] at (-0.05,3.25)
      {$t=0$: ignition (\texttt{startMotor})};
  \draw[station] (4.2,0) -- (4.2,2.95);
  \node[font=\sffamily\scriptsize, text=qrTeal, anchor=south west] at (4.1,2.98)
      {burnout};

  % recorded-step marks: filled = gate opens (rebuild), open = cache hit
  \node[font=\sffamily\scriptsize, anchor=east] at (-0.15,-0.55) {recorded steps:};
  \foreach \x in {0.4,0.9,...,4.0} { \node[cmdot] at (\x,-0.55) {}; }
  \foreach \x in {4.7,5.5,...,11.1} { \draw[ghost] (\x,-0.55) circle (1.7pt); }
  \node[font=\sffamily\scriptsize, text=qrRed, anchor=west] at (0.35,-0.95)
      {filled: gate opens, one rebuild};
  \node[font=\sffamily\scriptsize, text=qrGray, anchor=west] at (4.65,-0.95)
      {open: cache hit, zero rebuilds};

  % cadence braces
  \draw[decorate, decoration={brace, mirror, raise=2pt}, draw=qrRed, line width=0.6pt]
      (0,-1.3) -- (4.2,-1.3)
      node[midway, below=8pt, font=\sffamily\scriptsize, text=qrRed, align=center]
      {burn: $m$ falls between recorded steps,\\
       so \texttt{mNow} $\neq$ \texttt{builtAtMass\_} on every read};
  \draw[decorate, decoration={brace, mirror, raise=2pt}, draw=qrGreen, line width=0.6pt]
      (4.45,-1.3) -- (11.2,-1.3)
      node[midway, below=8pt, font=\sffamily\scriptsize, text=qrGreen, align=center]
      {post-burnout: the live sum repeats bit-for-bit,\\
       exact \texttt{==} holds and the cache freezes};
\end{tikzpicture}
