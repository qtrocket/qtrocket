% Geometry convention: +z = forward, part local frame [-L, 0] (FORE-plane origin),
% fractional stations, and the cone CM at hbar - L = -3L/4 (the (geom-center - CM)
% divergence the design eliminates). Consumed by section 02. Scale: 1 unit = 1 cm, L = 8.
\begin{tikzpicture}[x=1cm,y=1cm,>={Stealth[length=2.2mm]}]
  \def\L{8}      % length in cm
  \def\R{1.5}    % base radius in cm

  % --- the longitudinal axis: +z forward (right); the fore origin (tip) is at z = 0 ---
  \draw[axis] (-\L-1.3,0) -- (1.7,0) node[right,font=\small]{$+z$ (forward, toward nose tip)};

  % --- the cone (solid nose cone): tip (apex) at the origin z=0, wide base at z=-L ---
  \fill[qrBlue!10] (0,0) -- (-\L,\R) -- (-\L,-\R) -- cycle;
  \draw[part] (0,0) -- (-\L,\R) -- (-\L,-\R) -- cycle;

  % fore point (tip) at z = 0 -- the LOCAL ORIGIN
  \draw[station] (0,-\R-0.5) -- (0,\R+0.9);
  \node[align=center,font=\scriptsize,above] at (0,\R+0.9)
    {fore (tip)\\ \texttt{station01}=1,\ $r=0$};
  \node[font=\scriptsize,below,text=qrGray,align=center] at (0,-\R-0.5)
    {local origin\\ (fore plane)};

  % aft plane (base rim) at z = -L
  \draw[station] (-\L,-\R-0.5) -- (-\L,\R+0.9);
  \node[align=center,font=\scriptsize,above] at (-\L,\R+0.9)
    {aft plane (base rim)\\ \texttt{station01}=0,\ $r=$ baseRadius};

  % mid-length (geometric centre) at z = -L/2
  \draw[ghost] (-\L/2,-\R-0.2) -- (-\L/2,\R+0.2);
  \node[font=\scriptsize,below,text=qrGray,align=center] at (-\L/2,-\R-0.2)
    {$-L/2$\\ geometric centre};

  % centre of mass at z = hbar - L = -3L/4 (solid cone) -- near the wide base
  \node[cmdot] (cm) at (-3*\L/4,0) {};
  \node[font=\scriptsize,text=qrRed,align=center] (cmlbl) at (-3*\L/4,1.95)
    {CM $=\bar h - L = -3L/4$\\ \footnotesize(near the wide base)};
  \draw[qrRed,line width=0.4pt] (cmlbl.south) -- (-3*\L/4,0.06);

  % the divergence brace: (CM) -> (geom centre) = L/4
  \draw[decorate,decoration={brace,amplitude=4pt,raise=2pt},qrAmber]
    (-3*\L/4,-\R-1.4) -- (-\L/2,-\R-1.4)
    node[midway,below=4pt,font=\scriptsize,text=qrAmber,align=center]
    {$(\text{geom centre}-\text{CM})=L/4$\\ the offset the legacy CM-to-CM link bakes in};

  % a generic interior station s (e.g. s=0.7 -> z=(s-1)L)
  \draw[station] (-0.3*\L,-0.9) -- (-0.3*\L,0.9);
  \node[font=\scriptsize,below,text=qrTeal] at (-0.3*\L,-0.9) {\texttt{stationAt(s)}: $z=(s-1)\,L$};

  % radius profile annotation
  \node[font=\scriptsize,align=left,anchor=east,text=qrDeep] at (-\L*0.28,\R*0.60)
    {$\texttt{radiusOuterAt}(z)=\text{baseR}\,(-z/L)$};

  % occupancy span [-L, 0]
  \draw[dimline] (-\L,-\R-0.5) -- (0,-\R-0.5);
  \node[font=\scriptsize,fill=white,inner sep=1pt] at (-\L/2,-\R-0.5) {part occupies $z\in[-L,\,0]$};
