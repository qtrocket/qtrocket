% Activity diagram: resolvePlacements() -- placing one child of an already-placed
% parent (3-DOF today, 6-DOF-shaped). Consumed by section 04 (resolver).
\begin{tikzpicture}[node distance=6.5mm and 12mm]
  \node[startstop] (start) {DFS visit: child of a placed parent, link \texttt{m}};

  % p and c hold unbreakable typewriter calls (~51mm) -- give them room and put each
  % call on its own line so it never spills past the box edge.
  \node[act,below=of start,text width=60mm] (p)
    {$p \leftarrow$\\ \texttt{parent.stationAt(m.parentStation01)}\\
     \footnotesize $\{z, r_\text{outer}, r_\text{inner}\}$ in parent frame};
  \node[act,below=of p,text width=60mm] (c)
    {$c \leftarrow$\\ \texttt{child.stationAt(m.childStation01)}};

  \node[decision,below=of c] (seat) {seat kind?};

  % --- two seat branches, symmetric under the decision (10mm centre gap) ---
  \node[act,text width=36mm,anchor=north east] (abut) at ([xshift=-5mm,yshift=-9mm]seat.south)
    {\seat{Abut} / \seat{OnSurface}:\\ \texttt{signedGap} $= +$\,\texttt{m.gap}};
  \node[act,text width=36mm,anchor=north west] (nest) at ([xshift=5mm,yshift=-9mm]seat.south)
    {\seat{NestInBore}:\\ \texttt{signedGap} $= -$\,\texttt{m.gap}\\ \footnotesize(insertion depth, child seats aft)};

  % --- merge target: centred under the decision, 10mm below the TALLER branch (nest) ---
  \node[act,text width=58mm,anchor=north] (origin) at ($(seat.south |- nest.south) + (0,-10mm)$)
    {\texttt{childOriginZ} $= p.z + $\,\texttt{signedGap}$\,- c.z$\\ \footnotesize coaxial: $x=y=0$ (radial $r$ is a \emph{check}, not a placement)};
  \node[act,below=of origin,text width=58mm] (compose)
    {\texttt{childInParent} $=$ Pose\{(0,0,childOriginZ),\ m.childRot\}\\
     \texttt{childInRoot} $=$ \texttt{parentPose.compose(childInParent)}};
  \node[act,below=of compose,text width=58mm] (record) {record \texttt{Placed\{child, childInRoot\}} into \texttt{resolvedCache}};
  \node[decision,below=8mm of record] (seam) {Layer-1\\ radial seam ok?};
  \node[failnode,right=16mm of seam,text width=34mm] (diag) {append \texttt{OverlapDiagnostic};\\ \texttt{SolveResult.ok=false}};
  \node[startstop,below=8mm of seam] (recurse) {recurse into child's children, then return};

  \draw[flow] (start) -- (p);
  \draw[flow] (p) -- (c);
  \draw[flow] (c) -- (seat);
  \draw[flow] (seat) -| (abut);
  \draw[flow] (seat) -| (nest);

  % both branches drop to a junction 4mm ABOVE origin, then one arrow enters its top edge
  \coordinate (mrg) at ([yshift=4mm]origin.north);
  \draw[flow,-] (abut.south) |- (mrg);
  \draw[flow,-] (nest.south) |- (mrg);
  \draw[flow] (mrg) -- (origin.north);

  \draw[flow] (origin) -- (compose);
  \draw[flow] (compose) -- (record);
  \draw[flow] (record) -- (seam);
  \draw[flow] (seam) -- node[flowlbl]{no} (diag);
  \draw[flow] (seam) -- node[flowlbl]{yes} (recurse);
\end{tikzpicture}
