% A titled container box drawn behind a group of component nodes.

  \node[draw=#3,rounded corners=3pt,line width=0.6pt,fill=#3!5,
        fit=#1,inner sep=7pt,label={[anchor=north west,font=\sffamily\bfseries\small,
        text=#3,xshift=2pt,yshift=-1pt]north west:#2}] {};}
