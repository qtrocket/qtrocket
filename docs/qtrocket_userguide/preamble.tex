% ---- REPL transcripts: the console environment ------------------------------
% A verbatim block styled like the minted listings (no background: transcripts
% read differently from code), for qtrocket-cli sessions. commandchars makes
% \ { } active inside so the semantic line markers below can colorize the
% transcript; literal backslashes/braces are vanishingly rare in REPL output --
% if one appears, use \char`\\ / \char`\{ / \char`\}.
%
%   \begin{console}
%   \uin{loadmotors data/motors/AeroTech.rse}
%   \ok loadmotors: 154 motors from data/motors/AeroTech.rse
%   \uin{setmotor Q9999}
%   \err setmotor: 'Q9999' not found (use listmotors)
%   \end{console}
\DefineVerbatimEnvironment{console}{Verbatim}{
  fontsize=\footnotesize,
  breaklines=true,
  breakanywhere=true,
  commandchars=\\\{\},
  frame=leftline,
  framerule=0.4pt,
  framesep=8pt,
  rulecolor=\color{qrRule},
  formatcom=\color{qrInk}
}
% What the user types at the REPL (the CLI itself prints no prompt).

\newcommand{\err}{\textcolor{qrRed}{\textbf{ERR}}}
% WARN lines and muted banner/comment lines inside transcripts.
\newcommand{\warnline}[1]{\textcolor{qrAmber}{\textbf{#1}}}
\newcommand{\muted}[1]{\textcolor{qrGray}{#1}}

\usepackage{xstring}  % for breakable code-path citations

% Source citation:  \src{model/parts/Part.h}{326}  ->  model/parts/Part.h:326
% Paths break at '/' so long citations near a line end do not run into the margin.
\newcommand{\src}[2]{\textcolor{qrDeep}{\texttt{\small\StrSubstitute{#1}{/}{/\allowbreak}\textcolor{qrGray}{:}#2}}}
% File/path citation without line:  \srcf{data/motors/AeroTech.rse}
\newcommand{\srcf}[1]{\textcolor{qrDeep}{\texttt{\small\StrSubstitute{#1}{/}{/\allowbreak}}}}
% Caption-safe, non-breaking variants (xstring's \StrSubstitute is fragile in moving args):
\newcommand{\srccap}[2]{\textcolor{qrDeep}{\texttt{\small #1\textcolor{qrGray}{:}#2}}}
\newcommand{\srcfcap}[1]{\textcolor{qrDeep}{\texttt{\small #1}}}

% Fit content to \linewidth but never enlarge it beyond natural size.
